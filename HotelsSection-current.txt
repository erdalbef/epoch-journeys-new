"use client";

import {
  useMemo,
  useState,
} from "react";
import {
  Search,
  Star,
} from "lucide-react";
import { toast } from "sonner";

type HotelRow = {
  hotelName: string;
  destination: string;
  nights: number;
  singlePerPerson: number;
  doubleTwinPerPerson: number;
  triplePerPerson: number;
  stayType: "CORE" | "PRE" | "POST";

  supplierId?: string;
  supplierName?: string;
  supplierPreferred?: boolean;
  supplierServiceId?: string;
  supplierRateId?: string;
  supplierRateName?: string;
  supplierRateCurrency?: string;
  supplierRateValidFrom?: string;
  supplierRateValidTo?: string;
  supplierMealBasis?: string | null;
  supplierCityTaxIncluded?: boolean | null;
  supplierCityTaxAmount?: number | null;
  supplierPorterageIncluded?: boolean | null;
  supplierBeveragePackageIncluded?: boolean | null;
  supplierBeveragePackageDescription?: string | null;
};

type SupplierHotelResult = {
  supplier: {
    id: string;
    name: string;
    code: string | null;
    preferred: boolean;
    rating: number | null;
    country: string | null;
    city: string | null;
    defaultCurrency: string;
  };

  service: {
    id: string;
    name: string;
    description: string | null;
    country: string | null;
    city: string | null;
    notes: string | null;
  };

  rate: {
    id: string;
    name: string;
    description: string | null;
    validFrom: string;
    validTo: string;
    currency: string;
    amount: number;
    unit: string;
    roomType: string | null;
    mealBasis: string | null;
    minPax: number | null;
    maxPax: number | null;
    breakfastIncluded: boolean | null;
    dinnerIncluded: boolean | null;
    dinnerAmount: number | null;
    dinnerUnit: string | null;
    dinnerNotes: string | null;
    cityTaxIncluded: boolean;
    cityTaxAmount: number | null;
    cityTaxCurrency: string | null;
    cityTaxUnit: string | null;
    porterageAvailable: boolean | null;
    porterageIncluded: boolean | null;
    porterageAmount: number | null;
    porterageUnit: string | null;
    beveragePackageIncluded: boolean | null;
    beveragePackageDescription: string | null;
    beveragePackageAmount: number | null;
    beveragePackageUnit: string | null;
    beverageNotes: string | null;
    freePlaces: string | null;
    freePlaceRatio: number | null;
    freePlaceMax: number | null;
    freePlaceNotes: string | null;
    minimumRooms: number | null;
    minimumNights: number | null;
    rateRestrictions: string | null;
    singleSupplement: number | null;
    tripleReduction: number | null;
    cancellationTerms: string | null;
    paymentTerms: string | null;
    supplierReference: string | null;
    lastVerifiedAt: string | null;
    notes: string | null;
  };

  effectiveCost: {
    singlePerPersonPerNight: number | null;
    doubleTwinPerPersonPerNight: number | null;
    triplePerPersonPerNight: number | null;
    singleStatus: string;
    doubleTwinStatus: string;
    tripleStatus: string;
    currency: string;
  };
};

type Props = {
  hotels: HotelRow[];
  currency: string;

  startDate: string;
  endDate: string;
  payingPassengers: number;

  onAddHotel: () => void;
  onRemoveHotel: (index: number) => void;

  onUpdateHotel: (
    index: number,
    patch: Partial<HotelRow>,
  ) => void;

  hotelCostPerPerson: (
    row: HotelRow,
    occupancy:
      | "single"
      | "doubleTwin"
      | "triple",
  ) => number;

  formatMoney: (
    amount: number,
    currency?: string,
  ) => string;

  toNumber: (
    value: unknown,
  ) => number;
};

type SearchState = {
  loading: boolean;
  error: string;
  results: SupplierHotelResult[];
  searchedCity: string;
};

const rowBackgrounds = [
  "bg-amber-50/70 border-amber-200",
  "bg-emerald-50/60 border-emerald-200",
  "bg-white border-slate-200",
  "bg-blue-50/60 border-blue-200",
];

const rowHeaderBackgrounds = [
  "bg-amber-100/60 border-amber-200",
  "bg-emerald-100/50 border-emerald-200",
  "bg-slate-50 border-slate-200",
  "bg-blue-100/50 border-blue-200",
];

function getStayBadge(
  stayType: HotelRow["stayType"],
) {
  switch (stayType) {
    case "PRE":
      return {
        label: "PRE-STAY",
        className:
          "bg-amber-100 text-amber-800",
      };

    case "POST":
      return {
        label: "POST-STAY",
        className:
          "bg-purple-100 text-purple-800",
      };

    case "CORE":
    default:
      return {
        label: "CORE",
        className:
          "bg-blue-100 text-blue-800",
      };
  }
}

function displayDate(
  value: string,
) {
  const match =
    value.match(
      /^(\d{4})-(\d{2})-(\d{2})/,
    );

  if (!match) {
    return value;
  }

  return `${match[3]}/${match[2]}/${match[1]}`;
}

function rateUnitLabel(
  value: string,
) {
  return value
    .replaceAll("_", " ")
    .toLowerCase()
    .replace(
      /\b\w/g,
      (letter) =>
        letter.toUpperCase(),
    );
}

function yesNoIncluded(
  value: boolean | null,
) {
  if (value === true) {
    return "Included";
  }

  if (value === false) {
    return "Not included";
  }

  return "Not specified";
}

function completeEffectiveCost(
  result: SupplierHotelResult,
) {
  return (
    result.effectiveCost
      .singlePerPersonPerNight !==
      null &&
    result.effectiveCost
      .doubleTwinPerPersonPerNight !==
      null &&
    result.effectiveCost
      .triplePerPersonPerNight !==
      null &&
    result.effectiveCost
      .singleStatus ===
      "COMPLETE" &&
    result.effectiveCost
      .doubleTwinStatus ===
      "COMPLETE" &&
    result.effectiveCost
      .tripleStatus ===
      "COMPLETE"
  );
}

export default function HotelsSection({
  hotels,
  currency,
  startDate,
  endDate,
  payingPassengers,
  onAddHotel,
  onRemoveHotel,
  onUpdateHotel,
  hotelCostPerPerson,
  formatMoney,
  toNumber,
}: Props) {
  const [
    searchStates,
    setSearchStates,
  ] = useState<
    Record<
      number,
      SearchState
    >
  >({});

  const safePax =
    useMemo(
      () =>
        Math.max(
          Math.floor(
            toNumber(
              payingPassengers,
            ),
          ),
          1,
        ),
      [
        payingPassengers,
        toNumber,
      ],
    );

  async function searchSupplierHotels(
    index: number,
    destination: string,
  ) {
    const city =
      destination.trim();

    if (!city) {
      toast.error(
        "Enter a destination/city first.",
      );
      return;
    }

    setSearchStates(
      (prev) => ({
        ...prev,
        [index]: {
          loading: true,
          error: "",
          results:
            prev[index]
              ?.results ??
            [],
          searchedCity:
            city,
        },
      }),
    );

    try {
      const params =
        new URLSearchParams(
          {
            city,
            pax:
              String(
                safePax,
              ),
          },
        );

      if (
        startDate
      ) {
        params.set(
          "startDate",
          startDate,
        );
      }

      if (
        endDate
      ) {
        params.set(
          "endDate",
          endDate,
        );
      }

      const response =
        await fetch(
          `/api/admin/quotes/supplier-hotels?${params.toString()}`,
          {
            method:
              "GET",
            cache:
              "no-store",
          },
        );

      const data =
        (await response
          .json()
          .catch(
            () =>
              null,
          )) as {
          ok?: boolean;
          error?: string;
          hotels?: SupplierHotelResult[];
        } | null;

      if (
        !response.ok ||
        !data?.ok
      ) {
        throw new Error(
          data?.error ||
            "Unable to load supplier hotels.",
        );
      }

      const results =
        data.hotels ??
        [];

      setSearchStates(
        (prev) => ({
          ...prev,
          [index]: {
            loading: false,
            error: "",
            results,
            searchedCity:
              city,
          },
        }),
      );

      if (
        results.length ===
        0
      ) {
        toast.info(
          `No active hotel rates found for ${city}.`,
        );
      }
    } catch (error) {
      const message =
        error instanceof
          Error
          ? error.message
          : "Unable to load supplier hotels.";

      setSearchStates(
        (prev) => ({
          ...prev,
          [index]: {
            loading: false,
            error:
              message,
            results: [],
            searchedCity:
              city,
          },
        }),
      );

      toast.error(
        message,
      );
    }
  }

  function selectSupplierHotel(
    index: number,
    result: SupplierHotelResult,
  ) {
    if (
      result.rate.currency.toUpperCase() !==
      currency.toUpperCase()
    ) {
      toast.error(
        `This supplier rate is in ${result.rate.currency}, while the quote is in ${currency}. Currency conversion is not automatic.`,
      );

      return;
    }

    if (
      !completeEffectiveCost(
        result,
      )
    ) {
      toast.error(
        "This supplier rate does not have a complete comparable hotel cost. Review the supplier rate before using it.",
      );

      return;
    }

    onUpdateHotel(
      index,
      {
        hotelName:
          result.supplier
            .name,

        destination:
          result.service
            .city ||
          result.supplier
            .city ||
          hotels[index]
            ?.destination ||
          "",

        singlePerPerson:
          result
            .effectiveCost
            .singlePerPersonPerNight ??
          0,

        doubleTwinPerPerson:
          result
            .effectiveCost
            .doubleTwinPerPersonPerNight ??
          0,

        triplePerPerson:
          result
            .effectiveCost
            .triplePerPersonPerNight ??
          0,

        supplierId:
          result.supplier
            .id,

        supplierName:
          result.supplier
            .name,

        supplierPreferred:
          result.supplier
            .preferred,

        supplierServiceId:
          result.service
            .id,

        supplierRateId:
          result.rate.id,

        supplierRateName:
          result.rate.name,

        supplierRateCurrency:
          result.rate
            .currency,

        supplierRateValidFrom:
          result.rate
            .validFrom,

        supplierRateValidTo:
          result.rate
            .validTo,

        supplierMealBasis:
          result.rate
            .mealBasis,

        supplierCityTaxIncluded:
          result.rate
            .cityTaxIncluded,

        supplierCityTaxAmount:
          result.rate
            .cityTaxAmount,

        supplierPorterageIncluded:
          result.rate
            .porterageIncluded,

        supplierBeveragePackageIncluded:
          result.rate
            .beveragePackageIncluded,

        supplierBeveragePackageDescription:
          result.rate
            .beveragePackageDescription,
      },
    );

    toast.success(
      `${result.supplier.name} / ${result.rate.name} selected.`,
    );
  }

  return (
    <section className="overflow-hidden rounded-xl border bg-white">
      <div className="flex flex-col gap-3 bg-[#001F3F] px-5 py-4 text-white md:flex-row md:items-center md:justify-between">
        <div>
          <h2 className="text-lg font-semibold">
            Hotels
          </h2>

          <p className="mt-1 max-w-4xl text-xs text-slate-200">
            Enter the city, compare active supplier hotel rates, and select the hotel that best fits
            the group. Manual hotel and rate entry remains available.
          </p>
        </div>

        <button
          type="button"
          onClick={
            onAddHotel
          }
          className="rounded-md border border-white/30 bg-white/10 px-4 py-2 text-sm font-medium text-white transition hover:bg-white/20"
        >
          + Add Hotel
        </button>
      </div>

      <div className="space-y-5 p-5">
        {hotels.map(
          (
            row,
            index,
          ) => {
            const singleTotal =
              hotelCostPerPerson(
                row,
                "single",
              );

            const doubleTwinTotal =
              hotelCostPerPerson(
                row,
                "doubleTwin",
              );

            const tripleTotal =
              hotelCostPerPerson(
                row,
                "triple",
              );

            const colorIndex =
              index %
              rowBackgrounds.length;

            const rowBackground =
              rowBackgrounds[
                colorIndex
              ];

            const rowHeaderBackground =
              rowHeaderBackgrounds[
                colorIndex
              ];

            const stayBadge =
              getStayBadge(
                row.stayType,
              );

            const searchState =
              searchStates[
                index
              ];

            return (
              <div
                key={
                  index
                }
                className={`overflow-hidden rounded-xl border-2 shadow-sm ${rowBackground}`}
              >
                <div
                  className={`flex flex-col gap-3 border-b px-4 py-3 md:flex-row md:items-center md:justify-between ${rowHeaderBackground}`}
                >
                  <div>
                    <div className="flex flex-wrap items-center gap-2">
                      <span className="rounded-full bg-white/80 px-2.5 py-1 text-xs font-bold text-slate-700 shadow-sm">
                        Row{" "}
                        {index +
                          1}
                      </span>

                      <div className="font-semibold text-[#001F3F]">
                        {row.hotelName ||
                          row.destination ||
                          "Hotel"}
                      </div>

                      <span
                        className={`rounded-full px-2 py-1 text-[11px] font-semibold ${stayBadge.className}`}
                      >
                        {
                          stayBadge.label
                        }
                      </span>

                      {row.supplierPreferred && (
                        <span className="inline-flex items-center gap-1 rounded-full bg-amber-100 px-2 py-1 text-[11px] font-semibold text-amber-800">
                          <Star className="h-3 w-3 fill-current" />
                          Preferred
                        </span>
                      )}
                    </div>

                    <div className="mt-1 text-xs text-slate-500">
                      {row.stayType ===
                      "CORE"
                        ? "Included in the main tour costing"
                        : row.stayType ===
                            "PRE"
                          ? "Optional pre-stay accommodation"
                          : "Optional post-stay accommodation"}
                    </div>
                  </div>

                  <div className="flex flex-wrap gap-4 text-sm">
                    <div>
                      <span className="text-slate-500">
                        Single:{" "}
                      </span>

                      <span className="font-semibold">
                        {formatMoney(
                          singleTotal,
                          currency,
                        )}
                      </span>
                    </div>

                    <div>
                      <span className="text-slate-500">
                        Double/Twin:{" "}
                      </span>

                      <span className="font-semibold">
                        {formatMoney(
                          doubleTwinTotal,
                          currency,
                        )}
                      </span>
                    </div>

                    <div>
                      <span className="text-slate-500">
                        Triple:{" "}
                      </span>

                      <span className="font-semibold">
                        {formatMoney(
                          tripleTotal,
                          currency,
                        )}
                      </span>
                    </div>
                  </div>
                </div>

                <div className="space-y-5 p-4">
                  <div className="rounded-lg border border-slate-200 bg-white p-4">
                    <div className="flex flex-col gap-3 md:flex-row md:items-end">
                      <label className="block flex-1">
                        <span className="mb-1 block text-sm font-medium">
                          Destination / City
                        </span>

                        <input
                          className="w-full rounded-md border bg-white p-2"
                          value={
                            row.destination
                          }
                          onChange={(
                            event,
                          ) =>
                            onUpdateHotel(
                              index,
                              {
                                destination:
                                  event
                                    .target
                                    .value,
                              },
                            )
                          }
                          placeholder="e.g. Fatima"
                        />
                      </label>

                      <button
                        type="button"
                        onClick={() =>
                          searchSupplierHotels(
                            index,
                            row.destination,
                          )
                        }
                        disabled={
                          searchState?.loading
                        }
                        className="inline-flex h-10 items-center justify-center gap-2 rounded-md bg-[#001F3F] px-4 text-sm font-semibold text-white transition hover:bg-[#002d59] disabled:cursor-not-allowed disabled:opacity-50"
                      >
                        <Search className="h-4 w-4" />

                        {searchState?.loading
                          ? "Searching..."
                          : "Find Supplier Hotels"}
                      </button>
                    </div>

                    <p className="mt-2 text-xs text-slate-500">
                      Search uses the quote dates
                      {startDate
                        ? ` (${startDate}${endDate ? ` to ${endDate}` : ""})`
                        : ""}
                      {" "}and {safePax} paying pilgrims when available.
                    </p>

                    {searchState?.error && (
                      <p className="mt-3 rounded-md bg-red-50 px-3 py-2 text-sm text-red-700">
                        {
                          searchState.error
                        }
                      </p>
                    )}

                    {searchState &&
                      !searchState.loading &&
                      searchState.results.length >
                        0 && (
                        <div className="mt-4 space-y-3">
                          <div className="text-sm font-semibold text-slate-800">
                            Available supplier rates for{" "}
                            {
                              searchState.searchedCity
                            }
                          </div>

                          {searchState.results.map(
                            (
                              result,
                            ) => {
                              const complete =
                                completeEffectiveCost(
                                  result,
                                );

                              const sameCurrency =
                                result.rate.currency.toUpperCase() ===
                                currency.toUpperCase();

                              return (
                                <div
                                  key={`${result.supplier.id}-${result.service.id}-${result.rate.id}`}
                                  className="rounded-lg border border-slate-200 bg-slate-50 p-4"
                                >
                                  <div className="flex flex-col gap-3 lg:flex-row lg:items-start lg:justify-between">
                                    <div>
                                      <div className="flex flex-wrap items-center gap-2">
                                        <span className="font-semibold text-[#001F3F]">
                                          {
                                            result.supplier.name
                                          }
                                        </span>

                                        {result.supplier.preferred && (
                                          <span className="inline-flex items-center gap-1 rounded-full bg-amber-100 px-2 py-1 text-[11px] font-semibold text-amber-800">
                                            <Star className="h-3 w-3 fill-current" />
                                            Preferred
                                          </span>
                                        )}

                                        {result.supplier.rating !==
                                          null && (
                                          <span className="text-xs text-slate-500">
                                            Rating:{" "}
                                            {
                                              result.supplier.rating
                                            }
                                            /5
                                          </span>
                                        )}
                                      </div>

                                      <p className="mt-1 text-sm text-slate-600">
                                        {
                                          result.rate.name
                                        }
                                        {" · "}
                                        {
                                          result.rate.mealBasis ||
                                          "Meal basis not specified"
                                        }
                                        {" · "}
                                        {rateUnitLabel(
                                          result.rate.unit,
                                        )}
                                      </p>

                                      <p className="mt-1 text-xs text-slate-500">
                                        Valid{" "}
                                        {displayDate(
                                          result.rate.validFrom,
                                        )}
                                        {" - "}
                                        {displayDate(
                                          result.rate.validTo,
                                        )}
                                      </p>
                                    </div>

                                    <button
                                      type="button"
                                      disabled={
                                        !complete ||
                                        !sameCurrency
                                      }
                                      onClick={() =>
                                        selectSupplierHotel(
                                          index,
                                          result,
                                        )
                                      }
                                      className="rounded-md bg-[#8B0000] px-4 py-2 text-sm font-semibold text-white transition hover:bg-[#760000] disabled:cursor-not-allowed disabled:bg-slate-300"
                                    >
                                      Use This Rate
                                    </button>
                                  </div>

                                  <div className="mt-4 grid gap-3 sm:grid-cols-3">
                                    <RateBox
                                      label="Single"
                                      value={
                                        result
                                          .effectiveCost
                                          .singlePerPersonPerNight
                                      }
                                      status={
                                        result
                                          .effectiveCost
                                          .singleStatus
                                      }
                                      currency={
                                        result.rate.currency
                                      }
                                      formatMoney={
                                        formatMoney
                                      }
                                    />

                                    <RateBox
                                      label="Double/Twin"
                                      value={
                                        result
                                          .effectiveCost
                                          .doubleTwinPerPersonPerNight
                                      }
                                      status={
                                        result
                                          .effectiveCost
                                          .doubleTwinStatus
                                      }
                                      currency={
                                        result.rate.currency
                                      }
                                      formatMoney={
                                        formatMoney
                                      }
                                    />

                                    <RateBox
                                      label="Triple"
                                      value={
                                        result
                                          .effectiveCost
                                          .triplePerPersonPerNight
                                      }
                                      status={
                                        result
                                          .effectiveCost
                                          .tripleStatus
                                      }
                                      currency={
                                        result.rate.currency
                                      }
                                      formatMoney={
                                        formatMoney
                                      }
                                    />
                                  </div>

                                  <div className="mt-3 grid gap-2 text-xs text-slate-600 sm:grid-cols-2 lg:grid-cols-4">
                                    <Info
                                      label="Dinner"
                                      value={
                                        yesNoIncluded(
                                          result.rate
                                            .dinnerIncluded,
                                        )
                                      }
                                    />

                                    <Info
                                      label="City tax"
                                      value={
                                        result.rate
                                          .cityTaxIncluded
                                          ? "Included"
                                          : result.rate
                                                .cityTaxAmount !==
                                              null
                                            ? `${formatMoney(
                                                result.rate.cityTaxAmount,
                                                result.rate.cityTaxCurrency ||
                                                  result.rate.currency,
                                              )} · ${result.rate.cityTaxUnit ? rateUnitLabel(result.rate.cityTaxUnit) : "unit not specified"}`
                                            : "Not included / no amount"
                                      }
                                    />

                                    <Info
                                      label="Porterage"
                                      value={
                                        result.rate
                                          .porterageAvailable ===
                                        false
                                          ? "Not available"
                                          : yesNoIncluded(
                                              result.rate
                                                .porterageIncluded,
                                            )
                                      }
                                    />

                                    <Info
                                      label="Drinks"
                                      value={
                                        result.rate
                                          .beveragePackageIncluded
                                          ? result.rate
                                                .beveragePackageDescription ||
                                            "Included"
                                          : "Not included / not specified"
                                      }
                                    />
                                  </div>

                                  {!sameCurrency && (
                                    <p className="mt-3 text-xs font-medium text-amber-700">
                                      Rate currency is{" "}
                                      {
                                        result.rate.currency
                                      }
                                      , while this quote is{" "}
                                      {
                                        currency
                                      }
                                      . Automatic currency conversion is disabled.
                                    </p>
                                  )}

                                  {!complete && (
                                    <p className="mt-3 text-xs font-medium text-amber-700">
                                      Effective hotel cost is incomplete. Review the supplier rate before using it.
                                    </p>
                                  )}
                                </div>
                              );
                            },
                          )}
                        </div>
                      )}
                  </div>

                  <div className="grid gap-4 md:grid-cols-3">
                    <label className="block">
                      <span className="mb-1 block text-sm font-medium">
                        Stay Type
                      </span>

                      <select
                        className="w-full rounded-md border bg-white p-2"
                        value={
                          row.stayType
                        }
                        onChange={(
                          event,
                        ) =>
                          onUpdateHotel(
                            index,
                            {
                              stayType:
                                event
                                  .target
                                  .value as HotelRow["stayType"],
                            },
                          )
                        }
                      >
                        <option value="CORE">
                          Core Tour Hotel
                        </option>

                        <option value="PRE">
                          Pre-Stay Add-On
                        </option>

                        <option value="POST">
                          Post-Stay Add-On
                        </option>
                      </select>
                    </label>

                    <label className="block">
                      <span className="mb-1 block text-sm font-medium">
                        Hotel Name
                      </span>

                      <input
                        className="w-full rounded-md border bg-white p-2"
                        value={
                          row.hotelName
                        }
                        onChange={(
                          event,
                        ) =>
                          onUpdateHotel(
                            index,
                            {
                              hotelName:
                                event
                                  .target
                                  .value,
                            },
                          )
                        }
                        placeholder="e.g. Hotel Regina"
                      />
                    </label>

                    <label className="block">
                      <span className="mb-1 block text-sm font-medium">
                        Destination / City
                      </span>

                      <input
                        className="w-full rounded-md border bg-white p-2"
                        value={
                          row.destination
                        }
                        onChange={(
                          event,
                        ) =>
                          onUpdateHotel(
                            index,
                            {
                              destination:
                                event
                                  .target
                                  .value,
                            },
                          )
                        }
                        placeholder="e.g. Fatima"
                      />
                    </label>
                  </div>

                  {row.supplierRateId && (
                    <div className="rounded-lg border border-emerald-200 bg-emerald-50 p-4">
                      <div className="flex flex-wrap items-center gap-2">
                        <span className="text-sm font-semibold text-emerald-900">
                          Supplier rate selected:
                        </span>

                        <span className="text-sm text-emerald-800">
                          {row.supplierName ||
                            row.hotelName}
                          {" · "}
                          {row.supplierRateName ||
                            "Rate"}
                        </span>
                      </div>

                      <div className="mt-2 flex flex-wrap gap-x-5 gap-y-1 text-xs text-emerald-800">
                        {row.supplierRateCurrency && (
                          <span>
                            Currency:{" "}
                            {
                              row.supplierRateCurrency
                            }
                          </span>
                        )}

                        {row.supplierMealBasis && (
                          <span>
                            Meal basis:{" "}
                            {
                              row.supplierMealBasis
                            }
                          </span>
                        )}

                        {row.supplierRateValidFrom &&
                          row.supplierRateValidTo && (
                            <span>
                              Valid:{" "}
                              {displayDate(
                                row.supplierRateValidFrom,
                              )}
                              {" - "}
                              {displayDate(
                                row.supplierRateValidTo,
                              )}
                            </span>
                          )}
                      </div>

                      <p className="mt-2 text-xs text-emerald-700">
                        The supplier/rate reference is saved with the quote snapshot. The rate fields below remain editable.
                      </p>
                    </div>
                  )}

                  <div className="rounded-lg border bg-white/80 p-4">
                    <div className="grid gap-4 md:grid-cols-4">
                      <label className="block">
                        <span className="mb-1 block text-sm font-medium">
                          Nights
                        </span>

                        <input
                          type="number"
                          min="0"
                          step="1"
                          className="w-full rounded-md border bg-white p-2"
                          value={
                            row.nights
                          }
                          onChange={(
                            event,
                          ) =>
                            onUpdateHotel(
                              index,
                              {
                                nights:
                                  toNumber(
                                    event
                                      .target
                                      .value,
                                  ),
                              },
                            )
                          }
                        />
                      </label>

                      <label className="block">
                        <span className="mb-1 block text-sm font-medium">
                          Single / Person / Night
                        </span>

                        <input
                          type="number"
                          min="0"
                          step="0.01"
                          className="w-full rounded-md border bg-white p-2"
                          value={
                            row.singlePerPerson
                          }
                          onChange={(
                            event,
                          ) =>
                            onUpdateHotel(
                              index,
                              {
                                singlePerPerson:
                                  toNumber(
                                    event
                                      .target
                                      .value,
                                  ),
                              },
                            )
                          }
                        />
                      </label>

                      <label className="block">
                        <span className="mb-1 block text-sm font-medium">
                          Double/Twin / Person / Night
                        </span>

                        <input
                          type="number"
                          min="0"
                          step="0.01"
                          className="w-full rounded-md border bg-white p-2"
                          value={
                            row.doubleTwinPerPerson
                          }
                          onChange={(
                            event,
                          ) =>
                            onUpdateHotel(
                              index,
                              {
                                doubleTwinPerPerson:
                                  toNumber(
                                    event
                                      .target
                                      .value,
                                  ),
                              },
                            )
                          }
                        />
                      </label>

                      <label className="block">
                        <span className="mb-1 block text-sm font-medium">
                          Triple / Person / Night
                        </span>

                        <input
                          type="number"
                          min="0"
                          step="0.01"
                          className="w-full rounded-md border bg-white p-2"
                          value={
                            row.triplePerPerson
                          }
                          onChange={(
                            event,
                          ) =>
                            onUpdateHotel(
                              index,
                              {
                                triplePerPerson:
                                  toNumber(
                                    event
                                      .target
                                      .value,
                                  ),
                              },
                            )
                          }
                        />
                      </label>
                    </div>
                  </div>

                  <div className="grid gap-3 md:grid-cols-3">
                    <TotalBox
                      label="Single Hotel Cost / Person"
                      value={
                        singleTotal
                      }
                      currency={
                        currency
                      }
                      formatMoney={
                        formatMoney
                      }
                    />

                    <TotalBox
                      label="Double/Twin Hotel Cost / Person"
                      value={
                        doubleTwinTotal
                      }
                      currency={
                        currency
                      }
                      formatMoney={
                        formatMoney
                      }
                    />

                    <TotalBox
                      label="Triple Hotel Cost / Person"
                      value={
                        tripleTotal
                      }
                      currency={
                        currency
                      }
                      formatMoney={
                        formatMoney
                      }
                    />
                  </div>

                  {row.stayType !==
                    "CORE" && (
                    <div className="rounded-lg border border-amber-200 bg-amber-50 p-4">
                      <div className="text-sm font-semibold text-amber-900">
                        Optional{" "}
                        {row.stayType ===
                        "PRE"
                          ? "Pre-Stay"
                          : "Post-Stay"}{" "}
                        Accommodation
                      </div>

                      <p className="mt-1 text-sm text-amber-800">
                        These hotel rates will be shown separately in the proposal and will not increase the main tour NET rate.
                      </p>
                    </div>
                  )}

                  <div className="flex flex-col gap-3 border-t border-slate-300/70 pt-4 md:flex-row md:items-center md:justify-between">
                    <div className="text-sm text-slate-600">
                      {row.stayType ===
                      "CORE"
                        ? "Included in the main tour accommodation costing."
                        : `Separate optional ${row.stayType === "PRE" ? "pre-stay" : "post-stay"} rate.`}
                    </div>

                    <button
                      type="button"
                      onClick={() =>
                        onRemoveHotel(
                          index,
                        )
                      }
                      disabled={
                        hotels.length ===
                        1
                      }
                      className="rounded-md border border-red-200 bg-white px-4 py-2 text-sm font-medium text-red-700 transition hover:bg-red-50 disabled:cursor-not-allowed disabled:opacity-40"
                    >
                      Remove Hotel
                    </button>
                  </div>
                </div>
              </div>
            );
          },
        )}

        <div className="flex justify-center border-t pt-5">
          <button
            type="button"
            onClick={
              onAddHotel
            }
            className="rounded-lg border-2 border-dashed border-[#001F3F]/30 bg-slate-50 px-8 py-3 text-sm font-semibold text-[#001F3F] transition hover:border-[#001F3F]/60 hover:bg-blue-50"
          >
            + Add Another Hotel
          </button>
        </div>
      </div>
    </section>
  );
}

function RateBox({
  label,
  value,
  status,
  currency,
  formatMoney,
}: {
  label: string;
  value: number | null;
  status: string;
  currency: string;
  formatMoney: (
    amount: number,
    currency?: string,
  ) => string;
}) {
  return (
    <div className="rounded-md border bg-white p-3">
      <div className="text-xs font-medium uppercase tracking-wide text-slate-500">
        {label}
      </div>

      <div className="mt-1 text-base font-semibold text-[#001F3F]">
        {value ===
        null
          ? "Incomplete"
          : `${formatMoney(
              value,
              currency,
            )} pp/night`}
      </div>

      {status !==
        "COMPLETE" && (
        <div className="mt-1 text-xs text-amber-700">
          {status.replaceAll(
            "_",
            " ",
          )}
        </div>
      )}
    </div>
  );
}

function Info({
  label,
  value,
}: {
  label: string;
  value: string;
}) {
  return (
    <div className="rounded-md bg-white px-3 py-2">
      <span className="font-semibold">
        {label}:
      </span>{" "}
      {value}
    </div>
  );
}

function TotalBox({
  label,
  value,
  currency,
  formatMoney,
}: {
  label: string;
  value: number;
  currency: string;
  formatMoney: (
    amount: number,
    currency?: string,
  ) => string;
}) {
  return (
    <div className="rounded-md border bg-white p-3 text-sm">
      <div className="text-xs font-medium uppercase tracking-wide text-slate-500">
        {label}
      </div>

      <div className="mt-1 text-lg font-semibold text-[#001F3F]">
        {formatMoney(
          value,
          currency,
        )}
      </div>
    </div>
  );
}
