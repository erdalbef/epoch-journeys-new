import {
  Role,
  SupplierRateUnit,
  SupplierServiceType,
  SupplierStatus,
} from "@prisma/client";
import { getServerSession } from "next-auth";
import { NextResponse } from "next/server";

import { authOptions } from "@/lib/authOptions";
import { db } from "@/lib/db";

type EffectiveCostStatus =
  | "COMPLETE"
  | "INCOMPLETE"
  | "CURRENCY_MISMATCH"
  | "UNSUPPORTED_UNIT";

function clean(
  value: string | null,
) {
  return value?.trim() ?? "";
}

function optionalDate(
  value: string | null,
) {
  const cleaned =
    clean(value);

  if (!cleaned) {
    return null;
  }

  const parsed =
    new Date(
      `${cleaned}T00:00:00.000Z`,
    );

  return Number.isNaN(
    parsed.getTime(),
  )
    ? null
    : parsed;
}

function optionalPositiveInteger(
  value: string | null,
) {
  const cleaned =
    clean(value);

  if (!cleaned) {
    return null;
  }

  const parsed =
    Number(cleaned);

  if (
    !Number.isInteger(parsed) ||
    parsed <= 0
  ) {
    return null;
  }

  return parsed;
}

function decimalNumber(
  value:
    | {
        toString(): string;
      }
    | number
    | string
    | null
    | undefined,
) {
  if (
    value === null ||
    value === undefined
  ) {
    return null;
  }

  const parsed =
    Number(
      typeof value === "object"
        ? value.toString()
        : value,
    );

  return Number.isFinite(parsed)
    ? parsed
    : null;
}

function unitToPerPersonPerNight(
  amount: number,
  unit: SupplierRateUnit,
  occupancy: 1 | 2 | 3,
) {
  switch (unit) {
    case SupplierRateUnit.PER_PERSON_PER_NIGHT:
    case SupplierRateUnit.PER_PERSON:
    case SupplierRateUnit.PER_PERSON_PER_DAY:
    case SupplierRateUnit.PER_MEAL:
      return {
        value: amount,
        supported: true,
      };

    case SupplierRateUnit.PER_ROOM_PER_NIGHT:
    case SupplierRateUnit.PER_ROOM:
      return {
        value:
          amount /
          occupancy,
        supported: true,
      };

    default:
      return {
        value: 0,
        supported: false,
      };
  }
}

function componentCost({
  amount,
  unit,
  occupancy,
  included,
}: {
  amount: number | null;
  unit: SupplierRateUnit | null;
  occupancy: 1 | 2 | 3;
  included: boolean | null;
}) {
  if (included === true) {
    return {
      cost: 0,
      complete: true,
      supported: true,
    };
  }

  if (
    amount === null ||
    unit === null
  ) {
    return {
      cost: 0,
      complete: false,
      supported: true,
    };
  }

  const converted =
    unitToPerPersonPerNight(
      amount,
      unit,
      occupancy,
    );

  return {
    cost:
      converted.value,
    complete:
      converted.supported,
    supported:
      converted.supported,
  };
}

function roundMoney(
  value: number,
) {
  return Math.round(
    (value +
      Number.EPSILON) *
      100,
  ) / 100;
}

function effectiveHotelCost({
  occupancy,
  currency,
  baseAmount,
  baseUnit,
  dinnerIncluded,
  dinnerAmount,
  dinnerUnit,
  cityTaxIncluded,
  cityTaxAmount,
  cityTaxCurrency,
  cityTaxUnit,
  porterageAvailable,
  porterageIncluded,
  porterageAmount,
  porterageUnit,
  beveragePackageIncluded,
  beveragePackageAmount,
  beveragePackageUnit,
}: {
  occupancy: 1 | 2 | 3;
  currency: string;
  baseAmount: number;
  baseUnit: SupplierRateUnit;

  dinnerIncluded: boolean | null;
  dinnerAmount: number | null;
  dinnerUnit: SupplierRateUnit | null;

  cityTaxIncluded: boolean;
  cityTaxAmount: number | null;
  cityTaxCurrency: string | null;
  cityTaxUnit: SupplierRateUnit | null;

  porterageAvailable: boolean | null;
  porterageIncluded: boolean | null;
  porterageAmount: number | null;
  porterageUnit: SupplierRateUnit | null;

  beveragePackageIncluded: boolean | null;
  beveragePackageAmount: number | null;
  beveragePackageUnit: SupplierRateUnit | null;
}) {
  const base =
    unitToPerPersonPerNight(
      baseAmount,
      baseUnit,
      occupancy,
    );

  if (!base.supported) {
    return {
      amount: null,
      status:
        "UNSUPPORTED_UNIT" as EffectiveCostStatus,
    };
  }

  const dinner =
    componentCost({
      amount:
        dinnerAmount,
      unit:
        dinnerUnit,
      occupancy,
      included:
        dinnerIncluded,
    });

  const cityTaxCurrencyMismatch =
    cityTaxAmount !== null &&
    cityTaxIncluded !== true &&
    Boolean(
      cityTaxCurrency &&
        cityTaxCurrency.toUpperCase() !==
          currency.toUpperCase(),
    );

  if (
    cityTaxCurrencyMismatch
  ) {
    return {
      amount: null,
      status:
        "CURRENCY_MISMATCH" as EffectiveCostStatus,
    };
  }

  const cityTax =
    componentCost({
      amount:
        cityTaxAmount,
      unit:
        cityTaxUnit,
      occupancy,
      included:
        cityTaxIncluded,
    });

  const porterage =
    porterageAvailable ===
    false
      ? {
          cost: 0,
          complete: true,
          supported: true,
        }
      : componentCost({
          amount:
            porterageAmount,
          unit:
            porterageUnit,
          occupancy,
          included:
            porterageIncluded,
        });

  const beverage =
    componentCost({
      amount:
        beveragePackageAmount,
      unit:
        beveragePackageUnit,
      occupancy,
      included:
        beveragePackageIncluded,
    });

  const parts = [
    dinner,
    cityTax,
    porterage,
    beverage,
  ];

  if (
    parts.some(
      (part) =>
        !part.supported,
    )
  ) {
    return {
      amount: null,
      status:
        "UNSUPPORTED_UNIT" as EffectiveCostStatus,
    };
  }

  if (
    parts.some(
      (part) =>
        !part.complete,
    )
  ) {
    return {
      amount: null,
      status:
        "INCOMPLETE" as EffectiveCostStatus,
    };
  }

  return {
    amount:
      roundMoney(
        base.value +
          dinner.cost +
          cityTax.cost +
          porterage.cost +
          beverage.cost,
      ),
    status:
      "COMPLETE" as EffectiveCostStatus,
  };
}

export async function GET(
  request: Request,
) {
  try {
    const session =
      await getServerSession(
        authOptions,
      );

    if (
      !session?.user ||
      session.user.role !==
        Role.ADMIN
    ) {
      return NextResponse.json(
        {
          ok: false,
          error:
            "Unauthorized.",
        },
        {
          status: 401,
        },
      );
    }

    const {
      searchParams,
    } =
      new URL(
        request.url,
      );

    const city =
      clean(
        searchParams.get(
          "city",
        ),
      );

    const country =
      clean(
        searchParams.get(
          "country",
        ),
      );

    const startDate =
      optionalDate(
        searchParams.get(
          "startDate",
        ),
      );

    const endDate =
      optionalDate(
        searchParams.get(
          "endDate",
        ),
      );

    const pax =
      optionalPositiveInteger(
        searchParams.get(
          "pax",
        ),
      );

    if (!city) {
      return NextResponse.json(
        {
          ok: false,
          error:
            "City is required.",
        },
        {
          status: 400,
        },
      );
    }

    if (
      searchParams.get(
        "startDate",
      ) &&
      !startDate
    ) {
      return NextResponse.json(
        {
          ok: false,
          error:
            "Invalid startDate. Use YYYY-MM-DD.",
        },
        {
          status: 400,
        },
      );
    }

    if (
      searchParams.get(
        "endDate",
      ) &&
      !endDate
    ) {
      return NextResponse.json(
        {
          ok: false,
          error:
            "Invalid endDate. Use YYYY-MM-DD.",
        },
        {
          status: 400,
        },
      );
    }

    if (
      startDate &&
      endDate &&
      endDate <
        startDate
    ) {
      return NextResponse.json(
        {
          ok: false,
          error:
            "endDate cannot be earlier than startDate.",
        },
        {
          status: 400,
        },
      );
    }

    const coverageStart =
      startDate ??
      endDate;

    const coverageEnd =
      endDate ??
      startDate;

    const suppliers =
      await db.supplier.findMany(
        {
          where: {
            status:
              SupplierStatus.ACTIVE,

            services: {
              some: {
                isActive:
                  true,

                type:
                  SupplierServiceType.ACCOMMODATION,

                city: {
                  equals:
                    city,
                  mode:
                    "insensitive",
                },

                ...(country
                  ? {
                      country:
                        {
                          equals:
                            country,
                          mode:
                            "insensitive",
                        },
                    }
                  : {}),
              },
            },
          },

          select: {
            id: true,
            name: true,
            code: true,
            preferred: true,
            rating: true,
            country: true,
            city: true,
            defaultCurrency:
              true,

            services: {
              where: {
                isActive:
                  true,

                type:
                  SupplierServiceType.ACCOMMODATION,

                city: {
                  equals:
                    city,
                  mode:
                    "insensitive",
                },

                ...(country
                  ? {
                      country:
                        {
                          equals:
                            country,
                          mode:
                            "insensitive",
                        },
                    }
                  : {}),
              },

              orderBy: {
                name: "asc",
              },

              select: {
                id: true,
                name: true,
                description:
                  true,
                country: true,
                city: true,
                notes: true,

                rates: {
                  where: {
                    isActive:
                      true,

                    ...(pax
                      ? {
                          AND: [
                            {
                              OR: [
                                {
                                  minPax:
                                    null,
                                },
                                {
                                  minPax:
                                    {
                                      lte:
                                        pax,
                                    },
                                },
                              ],
                            },
                            {
                              OR: [
                                {
                                  maxPax:
                                    null,
                                },
                                {
                                  maxPax:
                                    {
                                      gte:
                                        pax,
                                    },
                                },
                              ],
                            },
                          ],
                        }
                      : {}),
                  },

                  orderBy: [
                    {
                      amount:
                        "asc",
                    },
                    {
                      validTo:
                        "desc",
                    },
                    {
                      name:
                        "asc",
                    },
                  ],

                  select: {
                    id: true,
                    supplierId:
                      true,
                    serviceId:
                      true,
                    name: true,
                    description:
                      true,

                    validFrom:
                      true,
                    validTo:
                      true,

                    currency:
                      true,
                    amount: true,
                    unit: true,

                    roomType:
                      true,
                    mealBasis:
                      true,

                    minPax: true,
                    maxPax: true,

                    breakfastIncluded:
                      true,

                    dinnerIncluded:
                      true,
                    dinnerAmount:
                      true,
                    dinnerUnit:
                      true,
                    dinnerNotes:
                      true,

                    cityTaxIncluded:
                      true,
                    cityTaxAmount:
                      true,
                    cityTaxCurrency:
                      true,
                    cityTaxUnit:
                      true,

                    porterageAvailable:
                      true,
                    porterageIncluded:
                      true,
                    porterageAmount:
                      true,
                    porterageUnit:
                      true,

                    beveragePackageIncluded:
                      true,
                    beveragePackageDescription:
                      true,
                    beveragePackageAmount:
                      true,
                    beveragePackageUnit:
                      true,
                    beverageNotes:
                      true,

                    freePlaces:
                      true,
                    freePlaceRatio:
                      true,
                    freePlaceMax:
                      true,
                    freePlaceNotes:
                      true,

                    minimumRooms:
                      true,
                    minimumNights:
                      true,
                    rateRestrictions:
                      true,

                    singleSupplement:
                      true,
                    tripleReduction:
                      true,

                    cancellationTerms:
                      true,
                    paymentTerms:
                      true,

                    supplierReference:
                      true,
                    lastVerifiedAt:
                      true,
                    notes: true,
                  },
                },
              },
            },
          },
        },
      );

    const hotels =
      suppliers
        .flatMap(
          (supplier) =>
            supplier.services.flatMap(
              (service) =>
                service.rates
                  .filter((rate) => {
                    const referenceStart =
                      coverageStart ??
                      new Date();

                    const referenceEnd =
                      coverageEnd ??
                      referenceStart;

                    const isValidForReference =
                      rate.validFrom <=
                        referenceStart &&
                      rate.validTo >=
                        referenceEnd;

                    const hasValidRate =
                      service.rates.some(
                        (candidate) =>
                          candidate.validFrom <=
                            referenceStart &&
                          candidate.validTo >=
                            referenceEnd,
                      );

                    if (hasValidRate) {
                      return isValidForReference;
                    }

                    const latestExpired =
                      service.rates
                        .filter(
                          (candidate) =>
                            candidate.validTo <
                            referenceStart,
                        )
                        .sort(
                          (a, b) =>
                            b.validTo.getTime() -
                            a.validTo.getTime(),
                        )[0];

                    return (
                      latestExpired?.id ===
                      rate.id
                    );
                  })
                  .map(
                  (rate) => {
                    const referenceStart =
                      coverageStart ??
                      new Date();

                    const referenceEnd =
                      coverageEnd ??
                      referenceStart;

                    const rateValidity =
                      rate.validFrom <=
                        referenceStart &&
                      rate.validTo >=
                        referenceEnd
                        ? "VALID_FOR_REQUEST"
                        : "EXPIRED_ESTIMATE";

                    const baseAmount =
                      decimalNumber(
                        rate.amount,
                      ) ?? 0;

                    const dinnerAmount =
                      decimalNumber(
                        rate.dinnerAmount,
                      );

                    const cityTaxAmount =
                      decimalNumber(
                        rate.cityTaxAmount,
                      );

                    const porterageAmount =
                      decimalNumber(
                        rate.porterageAmount,
                      );

                    const beveragePackageAmount =
                      decimalNumber(
                        rate.beveragePackageAmount,
                      );

                    const singleSupplement =
                      decimalNumber(
                        rate.singleSupplement,
                      );

                    const tripleReduction =
                      decimalNumber(
                        rate.tripleReduction,
                      );

                    const singleEffective =
                      effectiveHotelCost(
                        {
                          occupancy:
                            1,
                          currency:
                            rate.currency,
                          baseAmount,
                          baseUnit:
                            rate.unit,

                          dinnerIncluded:
                            rate.dinnerIncluded,
                          dinnerAmount,
                          dinnerUnit:
                            rate.dinnerUnit,

                          cityTaxIncluded:
                            rate.cityTaxIncluded,
                          cityTaxAmount,
                          cityTaxCurrency:
                            rate.cityTaxCurrency,
                          cityTaxUnit:
                            rate.cityTaxUnit,

                          porterageAvailable:
                            rate.porterageAvailable,
                          porterageIncluded:
                            rate.porterageIncluded,
                          porterageAmount,
                          porterageUnit:
                            rate.porterageUnit,

                          beveragePackageIncluded:
                            rate.beveragePackageIncluded,
                          beveragePackageAmount,
                          beveragePackageUnit:
                            rate.beveragePackageUnit,
                        },
                      );

                    const doubleEffective =
                      effectiveHotelCost(
                        {
                          occupancy:
                            2,
                          currency:
                            rate.currency,
                          baseAmount,
                          baseUnit:
                            rate.unit,

                          dinnerIncluded:
                            rate.dinnerIncluded,
                          dinnerAmount,
                          dinnerUnit:
                            rate.dinnerUnit,

                          cityTaxIncluded:
                            rate.cityTaxIncluded,
                          cityTaxAmount,
                          cityTaxCurrency:
                            rate.cityTaxCurrency,
                          cityTaxUnit:
                            rate.cityTaxUnit,

                          porterageAvailable:
                            rate.porterageAvailable,
                          porterageIncluded:
                            rate.porterageIncluded,
                          porterageAmount,
                          porterageUnit:
                            rate.porterageUnit,

                          beveragePackageIncluded:
                            rate.beveragePackageIncluded,
                          beveragePackageAmount,
                          beveragePackageUnit:
                            rate.beveragePackageUnit,
                        },
                      );

                    const tripleEffective =
                      effectiveHotelCost(
                        {
                          occupancy:
                            3,
                          currency:
                            rate.currency,
                          baseAmount,
                          baseUnit:
                            rate.unit,

                          dinnerIncluded:
                            rate.dinnerIncluded,
                          dinnerAmount,
                          dinnerUnit:
                            rate.dinnerUnit,

                          cityTaxIncluded:
                            rate.cityTaxIncluded,
                          cityTaxAmount,
                          cityTaxCurrency:
                            rate.cityTaxCurrency,
                          cityTaxUnit:
                            rate.cityTaxUnit,

                          porterageAvailable:
                            rate.porterageAvailable,
                          porterageIncluded:
                            rate.porterageIncluded,
                          porterageAmount,
                          porterageUnit:
                            rate.porterageUnit,

                          beveragePackageIncluded:
                            rate.beveragePackageIncluded,
                          beveragePackageAmount,
                          beveragePackageUnit:
                            rate.beveragePackageUnit,
                        },
                      );

                    const adjustedSingleEffective =
                      singleEffective.amount === null
                        ? singleEffective
                        : {
                            ...singleEffective,
                            amount: roundMoney(
                              singleEffective.amount +
                                (singleSupplement ?? 0),
                            ),
                          };

                    const adjustedTripleEffective =
                      tripleEffective.amount === null
                        ? tripleEffective
                        : {
                            ...tripleEffective,
                            amount: roundMoney(
                              Math.max(
                                0,
                                tripleEffective.amount -
                                  (tripleReduction ?? 0),
                              ),
                            ),
                          };

                    return {
                      supplier: {
                        id:
                          supplier.id,
                        name:
                          supplier.name,
                        code:
                          supplier.code,
                        preferred:
                          supplier.preferred,
                        rating:
                          supplier.rating,
                        country:
                          supplier.country,
                        city:
                          supplier.city,
                        defaultCurrency:
                          supplier.defaultCurrency,
                      },

                      service: {
                        id:
                          service.id,
                        name:
                          service.name,
                        description:
                          service.description,
                        country:
                          service.country,
                        city:
                          service.city,
                        notes:
                          service.notes,
                      },

                      rate: {
                        id:
                          rate.id,
                        name:
                          rate.name,
                        description:
                          rate.description,

                        validFrom:
                          rate.validFrom.toISOString(),
                        validTo:
                          rate.validTo.toISOString(),

                        currency:
                          rate.currency,
                        amount:
                          baseAmount,
                        unit:
                          rate.unit,

                        roomType:
                          rate.roomType,
                        mealBasis:
                          rate.mealBasis,

                        minPax:
                          rate.minPax,
                        maxPax:
                          rate.maxPax,

                        breakfastIncluded:
                          rate.breakfastIncluded,

                        dinnerIncluded:
                          rate.dinnerIncluded,
                        dinnerAmount,
                        dinnerUnit:
                          rate.dinnerUnit,
                        dinnerNotes:
                          rate.dinnerNotes,

                        cityTaxIncluded:
                          rate.cityTaxIncluded,
                        cityTaxAmount,
                        cityTaxCurrency:
                          rate.cityTaxCurrency,
                        cityTaxUnit:
                          rate.cityTaxUnit,

                        porterageAvailable:
                          rate.porterageAvailable,
                        porterageIncluded:
                          rate.porterageIncluded,
                        porterageAmount,
                        porterageUnit:
                          rate.porterageUnit,

                        beveragePackageIncluded:
                          rate.beveragePackageIncluded,
                        beveragePackageDescription:
                          rate.beveragePackageDescription,
                        beveragePackageAmount,
                        beveragePackageUnit:
                          rate.beveragePackageUnit,
                        beverageNotes:
                          rate.beverageNotes,

                        freePlaces:
                          rate.freePlaces,
                        freePlaceRatio:
                          rate.freePlaceRatio,
                        freePlaceMax:
                          rate.freePlaceMax,
                        freePlaceNotes:
                          rate.freePlaceNotes,

                        minimumRooms:
                          rate.minimumRooms,
                        minimumNights:
                          rate.minimumNights,
                        rateRestrictions:
                          rate.rateRestrictions,

                        singleSupplement,
                        tripleReduction,

                        cancellationTerms:
                          rate.cancellationTerms,
                        paymentTerms:
                          rate.paymentTerms,

                        supplierReference:
                          rate.supplierReference,

                        lastVerifiedAt:
                          rate.lastVerifiedAt
                            ? rate.lastVerifiedAt.toISOString()
                            : null,

                        notes:
                          rate.notes,
                      },

                      rateValidity,

                      requiresReconfirmation:
                        rateValidity ===
                        "EXPIRED_ESTIMATE",

                      effectiveCost: {
                        singlePerPersonPerNight:
                          adjustedSingleEffective.amount,

                        doubleTwinPerPersonPerNight:
                          doubleEffective.amount,

                        triplePerPersonPerNight:
                          adjustedTripleEffective.amount,

                        singleStatus:
                          adjustedSingleEffective.status,

                        doubleTwinStatus:
                          doubleEffective.status,

                        tripleStatus:
                          adjustedTripleEffective.status,

                        currency:
                          rate.currency,
                      },
                    };
                  },
                ),
            ),
        )
        .sort(
          (a, b) => {
            if (
              a.rateValidity !==
              b.rateValidity
            ) {
              return a.rateValidity ===
                "VALID_FOR_REQUEST"
                ? -1
                : 1;
            }

            if (
              a.supplier.preferred !==
              b.supplier.preferred
            ) {
              return a.supplier.preferred
                ? -1
                : 1;
            }

            const aCost =
              a.effectiveCost
                .doubleTwinPerPersonPerNight ??
              Number.POSITIVE_INFINITY;

            const bCost =
              b.effectiveCost
                .doubleTwinPerPersonPerNight ??
              Number.POSITIVE_INFINITY;

            if (
              aCost !==
              bCost
            ) {
              return (
                aCost -
                bCost
              );
            }

            const supplierCompare =
              a.supplier.name.localeCompare(
                b.supplier.name,
              );

            if (
              supplierCompare !==
              0
            ) {
              return supplierCompare;
            }

            return a.rate.name.localeCompare(
              b.rate.name,
            );
          },
        );

    return NextResponse.json(
      {
        ok: true,

        query: {
          city,
          country:
            country || null,

          startDate:
            startDate
              ? startDate.toISOString()
              : null,

          endDate:
            endDate
              ? endDate.toISOString()
              : null,

          pax,
        },

        hotels,
      },
      {
        status: 200,
      },
    );
  } catch (error) {
    console.error(
      "QUOTE_SUPPLIER_HOTELS_GET_ERROR",
      error,
    );

    return NextResponse.json(
      {
        ok: false,

        error:
          error instanceof
          Error
            ? error.message
            : "Could not load supplier hotels.",
      },
      {
        status: 500,
      },
    );
  }
}
